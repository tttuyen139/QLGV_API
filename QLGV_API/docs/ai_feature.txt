    da_dat, _ = ket_qua_hoc_tap(sinh_vien)

    if lhp.dang_ky.count() >= lhp.si_so_toi_da:

        sinh_vien=sinh_vien, lop_hoc_phan__hoc_ky=lhp.hoc_ky, lop_hoc_phan__nam_hoc=lhp.nam_hoc
    ).select_related("lop_hoc_phan")
    for dk in cung_ky:
        k = dk.lop_hoc_phan
        if k.mon_hoc_id == lhp.mon_hoc_id:
            loi.append(f"{lhp.ma_lhp}: đã đăng ký môn này (lớp {k.ma_lhp}).")
        elif lhp.thu and k.thu == lhp.thu and k.ca == lhp.ca:
            loi.append(f"{lhp.ma_lhp}: trùng lịch với lớp {k.ma_lhp}.")

    if loi:
        raise ValidationError(loi)

"""Xếp lịch học và phòng bằng lập trình ràng buộc (Google OR-Tools CP-SAT)."""
from collections import defaultdict
from itertools import combinations

from django.db import transaction
from ortools.sat.python import cp_model

from ..models import CA_CHOICES, THU_CHOICES, DangKyHocPhan, LopHocPhan, Phong

KHUNG_GIO = [(thu, ca) for thu, _ in THU_CHOICES for ca, _ in CA_CHOICES]


def _phat(thu, ca):
    """Ràng buộc mềm: hạn chế xếp thứ 7 và ca cuối ngày."""
    return (5 if thu == 7 else 0) + (3 if ca == 4 else 0)


def _cap_xung_dot(ds_lhp):
    """Các cặp lớp học phần có chung sinh viên đã đăng ký → không được trùng giờ."""
    theo_sv = defaultdict(set)
    rows = DangKyHocPhan.objects.filter(lop_hoc_phan__in=ds_lhp).values_list(
        "sinh_vien_id", "lop_hoc_phan_id")
    for sv_id, lhp_id in rows:
        theo_sv[sv_id].add(lhp_id)
    cap = set()
    for ds in theo_sv.values():
        cap.update(combinations(sorted(ds), 2))
    return cap


def xep_lich(hoc_ky, nam_hoc, gioi_han_giay=30):
    ds_lhp = list(LopHocPhan.objects.filter(hoc_ky=hoc_ky, nam_hoc=nam_hoc)
                  .select_related("mon_hoc", "giang_vien"))
    ds_phong = list(Phong.objects.all())
    if not ds_lhp:
        return {"thanh_cong": False, "thong_bao": "Học kỳ này chưa có lớp học phần.", "lich": []}

    model = cp_model.CpModel()
    bien = {}                               # (lhp_id, khung, phong_id) -> biến 0/1
    theo_lhp = defaultdict(list)
    theo_lhp_khung = defaultdict(list)
    theo_phong_khung = defaultdict(list)
    theo_gv_khung = defaultdict(list)
    chi_phi = []

    for l in ds_lhp:
        phong_hop_le = [p for p in ds_phong if p.suc_chua >= l.si_so_toi_da]
        if not phong_hop_le:
            return {"thanh_cong": False, "lich": [],
                    "thong_bao": f"Không có phòng đủ {l.si_so_toi_da} chỗ cho lớp {l.ma_lhp}."}
        for k, (thu, ca) in enumerate(KHUNG_GIO):
            for p in phong_hop_le:
                v = model.new_bool_var(f"x_{l.id}_{k}_{p.id}")
                bien[l.id, k, p.id] = v
                theo_lhp[l.id].append(v)
                theo_lhp_khung[l.id, k].append(v)
                theo_phong_khung[p.id, k].append(v)
                theo_gv_khung[l.giang_vien_id, k].append(v)
                lang_phi = (p.suc_chua - l.si_so_toi_da) // 10
                chi_phi.append((_phat(thu, ca) * 10 + lang_phi) * v)

    # Ràng buộc cứng
    for ds in theo_lhp.values():
        model.add_exactly_one(ds)        # mỗi lớp đúng 1 khung giờ + 1 phòng
    for ds in theo_phong_khung.values():
        model.add_at_most_one(ds)        # 1 phòng không chứa 2 lớp cùng lúc
    for ds in theo_gv_khung.values():
        model.add_at_most_one(ds)        # giảng viên không dạy 2 lớp cùng lúc
    for l1, l2 in _cap_xung_dot(ds_lhp):
        for k in range(len(KHUNG_GIO)):
            model.add_at_most_one(theo_lhp_khung[l1, k] + theo_lhp_khung[l2, k])

    model.minimize(sum(chi_phi))         # ràng buộc mềm: ít phạt, ít lãng phí chỗ

    solver = cp_model.CpSolver()
    solver.parameters.max_time_in_seconds = gioi_han_giay
    trang_thai = solver.solve(model)
    if trang_thai not in (cp_model.OPTIMAL, cp_model.FEASIBLE):
        return {"thanh_cong": False, "lich": [],
                "thong_bao": "Không tìm được lịch thỏa mãn (thiếu phòng hoặc giảng viên quá tải)."}

    lhp_theo_id = {l.id: l for l in ds_lhp}
    phong_theo_id = {p.id: p for p in ds_phong}
    lich = []
    for (l_id, k, p_id), v in bien.items():
        if solver.value(v):
            l, (thu, ca) = lhp_theo_id[l_id], KHUNG_GIO[k]
            lich.append({
                "lop_hoc_phan": l_id, "ma_lhp": l.ma_lhp, "ten_mon": l.mon_hoc.ten_mon,
                "giang_vien": l.giang_vien.ho_ten, "thu": thu, "ca": ca,
                "phong_id": p_id, "phong": phong_theo_id[p_id].ma_phong,
            })
    lich.sort(key=lambda d: (d["thu"], d["ca"], d["phong"]))
    return {"thanh_cong": True, "toi_uu": trang_thai == cp_model.OPTIMAL,
            "thong_bao": f"Đã xếp {len(lich)} lớp học phần.", "lich": lich}

"""Chọn tổ hợp lớp học phần tối ưu cho sinh viên bằng CP-SAT."""
from collections import defaultdict

from django.db.models import Count
from ortools.sat.python import cp_model

from ..models import LopHocPhan
from ..services import ket_qua_hoc_tap


def goi_y(sinh_vien, hoc_ky, nam_hoc, so_thich, gioi_han_giay=10):
    tc_min = so_thich.get("tin_chi_min", 12)
    tc_max = so_thich.get("tin_chi_max", 20)
    thu_tranh = set(so_thich.get("thu_tranh", []))
    ca_uu_tien = set(so_thich.get("ca_uu_tien", []))
    mon_uu_tien = set(so_thich.get("ma_mon_uu_tien", []))

    da_dat, chua_dat = ket_qua_hoc_tap(sinh_vien)
    da_dk = list(LopHocPhan.objects.filter(dang_ky__sinh_vien=sinh_vien, hoc_ky=hoc_ky,
                                           nam_hoc=nam_hoc).select_related("mon_hoc"))
    lich_ban = {(l.thu, l.ca) for l in da_dk if l.thu}
    mon_da_dk = {l.mon_hoc_id for l in da_dk}
    tc_da_dk = sum(l.mon_hoc.so_tin_chi for l in da_dk)

    if tc_da_dk >= tc_max:
        return {"de_xuat": [], "tong_tin_chi": tc_da_dk,
                "ghi_chu": "Bạn đã đăng ký đủ số tín chỉ tối đa."}

    # Lọc ứng viên theo ràng buộc cứng
    ung_vien = []
    qs = (LopHocPhan.objects.filter(hoc_ky=hoc_ky, nam_hoc=nam_hoc, thu__isnull=False)
          .select_related("mon_hoc", "giang_vien", "phong").annotate(so_dk=Count("dang_ky")))
    for l in qs:
        m = l.mon_hoc
        if (l.so_dk >= l.si_so_toi_da or m.id in da_dat or m.id in mon_da_dk
                or (m.mon_tien_quyet_id and m.mon_tien_quyet_id not in da_dat)
                or (l.thu, l.ca) in lich_ban or l.thu in thu_tranh):
            continue
        ung_vien.append(l)

    if not ung_vien:
        return {"de_xuat": [], "tong_tin_chi": tc_da_dk,
                "ghi_chu": "Không còn lớp học phần nào phù hợp với yêu cầu."}

    def diem(l):
        d = l.mon_hoc.so_tin_chi * 10
        if l.mon_hoc_id in chua_dat:
            d += 30                       # ưu tiên học lại môn chưa đạt
        if l.mon_hoc.ma_mon in mon_uu_tien:
            d += 20
        if l.ca in ca_uu_tien:
            d += 5
        return d

    def giai(tc_toi_thieu):
        model = cp_model.CpModel()
        y = {l.id: model.new_bool_var(f"y_{l.id}") for l in ung_vien}
        theo_mon, theo_khung = defaultdict(list), defaultdict(list)
        for l in ung_vien:
            theo_mon[l.mon_hoc_id].append(y[l.id])
            theo_khung[l.thu, l.ca].append(y[l.id])
        for ds in [*theo_mon.values(), *theo_khung.values()]:
            model.add_at_most_one(ds)     # mỗi môn 1 lớp, mỗi khung giờ 1 lớp
        tong = sum(l.mon_hoc.so_tin_chi * y[l.id] for l in ung_vien)
        model.add(tc_da_dk + tong >= tc_toi_thieu)
        model.add(tc_da_dk + tong <= tc_max)
        model.maximize(sum(diem(l) * y[l.id] for l in ung_vien))

        solver = cp_model.CpSolver()
        solver.parameters.max_time_in_seconds = gioi_han_giay
        if solver.solve(model) in (cp_model.OPTIMAL, cp_model.FEASIBLE):
            return [l for l in ung_vien if solver.value(y[l.id])]
        return None

    ghi_chu = ""
    chon = giai(tc_min)
    if chon is None:
        chon = giai(0) or []
        ghi_chu = f"Không đủ lớp phù hợp để đạt {tc_min} tín chỉ; đây là phương án tốt nhất có thể."

    de_xuat = [{
        "lop_hoc_phan": l.id, "ma_lhp": l.ma_lhp, "ma_mon": l.mon_hoc.ma_mon,
        "ten_mon": l.mon_hoc.ten_mon, "so_tin_chi": l.mon_hoc.so_tin_chi,
        "giang_vien": l.giang_vien.ho_ten, "thu": l.thu, "ca": l.ca,
        "phong": l.phong.ma_phong if l.phong else None,
        "hoc_lai": l.mon_hoc_id in chua_dat,
    } for l in sorted(chon, key=lambda l: (l.thu, l.ca))]
    return {"de_xuat": de_xuat,
            "tong_tin_chi": tc_da_dk + sum(d["so_tin_chi"] for d in de_xuat),
            "ghi_chu": ghi_chu}

import json
import logging

from django.conf import settings

log = logging.getLogger(__name__)

MAC_DINH = {"tin_chi_min": 12, "tin_chi_max": 20, "thu_tranh": [],
            "ca_uu_tien": [], "ma_mon_uu_tien": []}

CONG_CU = {
    "name": "ghi_nhan_so_thich",
    "description": "Ghi nhận sở thích đăng ký học phần của sinh viên.",
    "input_schema": {
        "type": "object",
        "properties": {
            "tin_chi_min": {"type": "integer", "description": "Số tín chỉ tối thiểu"},
            "tin_chi_max": {"type": "integer", "description": "Số tín chỉ tối đa"},
            "thu_tranh": {"type": "array", "items": {"type": "integer", "minimum": 2, "maximum": 7},
                          "description": "Các thứ muốn tránh (2 = Thứ 2 ... 7 = Thứ 7)"},
            "ca_uu_tien": {"type": "array", "items": {"type": "integer", "minimum": 1, "maximum": 4},
                           "description": "Ca muốn học: 1, 2 = buổi sáng; 3, 4 = buổi chiều"},
            "ma_mon_uu_tien": {"type": "array", "items": {"type": "string"},
                               "description": "Mã môn muốn ưu tiên, chỉ chọn từ danh sách được cung cấp"},
        },
        "required": ["tin_chi_min", "tin_chi_max", "thu_tranh", "ca_uu_tien", "ma_mon_uu_tien"],
    },
}


def _client():
    if not settings.ANTHROPIC_API_KEY:
        return None
    import anthropic
    return anthropic.Anthropic(api_key=settings.ANTHROPIC_API_KEY)


def phan_tich_yeu_cau(yeu_cau, ds_mon):
    """Trả về (sở thích, có dùng AI hay không)."""
    client = _client()
    if client is None or not yeu_cau:
        return dict(MAC_DINH), False
    danh_sach = "\n".join(f"- {m.ma_mon}: {m.ten_mon}" for m in ds_mon)
    try:
        resp = client.messages.create(
            model=settings.AI_MODEL,
            max_tokens=500,
            system=("Bạn là trợ lý đăng ký học phần. Trích xuất sở thích của sinh viên. "
                    "Trường nào sinh viên không nhắc đến thì dùng mặc định: "
                    "tin_chi_min=12, tin_chi_max=20, các danh sách để rỗng."),
            tools=[CONG_CU],
            tool_choice={"type": "tool", "name": "ghi_nhan_so_thich"},
            messages=[{"role": "user", "content":
                       f"Các môn mở trong kỳ:\n{danh_sach}\n\nYêu cầu của sinh viên: {yeu_cau}"}],
        )
        for block in resp.content:
            if block.type == "tool_use":
                return {**MAC_DINH, **block.input}, True
    except Exception:
        log.exception("Lỗi gọi Claude API")
    return dict(MAC_DINH), False


def giai_thich(yeu_cau, ket_qua):
    mac_dinh = (f"Đề xuất {len(ket_qua['de_xuat'])} lớp học phần, tổng {ket_qua['tong_tin_chi']} "
                f"tín chỉ, không trùng lịch và đã đủ điều kiện tiên quyết. {ket_qua['ghi_chu']}").strip()
    client = _client()
    if client is None:
        return mac_dinh
    try:
        resp = client.messages.create(
            model=settings.AI_MODEL,
            max_tokens=600,
            system=("Bạn là cố vấn học tập. Giải thích ngắn gọn bằng tiếng Việt (tối đa 5 câu) "
                    "vì sao danh sách học phần phù hợp với yêu cầu. Chỉ dựa trên dữ liệu được cung cấp."),
            messages=[{"role": "user", "content":
                       f"Yêu cầu: {yeu_cau}\nKết quả: {json.dumps(ket_qua, ensure_ascii=False)}"}],
        )
        return "".join(b.text for b in resp.content if b.type == "text") or mac_dinh
    except Exception:
        log.exception("Lỗi gọi Claude API")
        return mac_dinh

def _lay_hoc_ky(data):
    hoc_ky, nam_hoc = data.get("hoc_ky"), data.get("nam_hoc")
    if not hoc_ky or not nam_hoc:
        raise ValidationError("Cần truyền hoc_ky và nam_hoc.")
    return hoc_ky, nam_hoc


class AIGoiYDangKyView(APIView):
    """SV gõ yêu cầu tiếng Việt → AI đề xuất tổ hợp lớp học phần (CHƯA ghi CSDL)."""

    def post(self, request):
        sv = getattr(request.user, "sinh_vien", None)
        if sv is None:
            raise PermissionDenied("Chức năng dành cho sinh viên.")
        hoc_ky, nam_hoc = _lay_hoc_ky(request.data)
        yeu_cau = request.data.get("yeu_cau", "")

        ds_mon = MonHoc.objects.filter(lop_hoc_phan__hoc_ky=hoc_ky,
                                       lop_hoc_phan__nam_hoc=nam_hoc).distinct()
        so_thich, co_dung_ai = phan_tich_yeu_cau(yeu_cau, ds_mon)
        ket_qua = goi_y(sv, hoc_ky, nam_hoc, so_thich)
        ket_qua["giai_thich"] = giai_thich(yeu_cau, ket_qua)
        return Response({"so_thich": so_thich, "co_dung_ai": co_dung_ai, **ket_qua})


class AIXacNhanDangKyView(APIView):
    """SV xác nhận đề xuất → kiểm tra lại toàn bộ ràng buộc rồi mới ghi CSDL."""

    def post(self, request):
        sv = getattr(request.user, "sinh_vien", None)
        if sv is None:
            raise PermissionDenied("Chức năng dành cho sinh viên.")
        ids = request.data.get("lop_hoc_phan", [])
        ds = list(LopHocPhan.objects.filter(id__in=ids).select_related("mon_hoc"))
        if not ds:
            raise ValidationError("Danh sách lớp học phần trống.")
        with transaction.atomic():           # lỗi 1 lớp → hủy toàn bộ
            for lhp in ds:
                kiem_tra_dang_ky(sv, lhp)
                DangKyHocPhan.objects.create(sinh_vien=sv, lop_hoc_phan=lhp)
        return Response({"da_dang_ky": [l.ma_lhp for l in ds]}, status=201)


class AIXepLichView(APIView):
    """Quản trị viên chạy AI xếp lịch + phòng. luu=false để xem trước."""
    permission_classes = [LaQuanTri]

    def post(self, request):
        hoc_ky, nam_hoc = _lay_hoc_ky(request.data)
        ket_qua = xep_lich(hoc_ky, nam_hoc)
        if ket_qua["thanh_cong"] and request.data.get("luu"):
            luu_lich(ket_qua["lich"])
            ket_qua["da_luu"] = True
        return Response(ket_qua)

6. Gợi ý trình bày trong báo cáo
Bài toán thỏa mãn ràng buộc (CSP): phân biệt ràng buộc cứng (không trùng phòng, giảng viên, sinh viên; phòng đủ sức chứa; đạt môn tiên quyết) và ràng buộc mềm (tránh thứ 7, ca muộn, phòng quá rộng; ưu tiên môn học lại). Đây là kỹ thuật AI cổ điển dựa trên tìm kiếm có ràng buộc.
Mô hình ngôn ngữ lớn (LLM): dùng tool use để chuyển câu tiếng Việt thành dữ liệu có cấu trúc. LLM không truy cập trực tiếp CSDL và không tự quyết định đăng ký.
Vị trí trong MVC: thư mục ai/ và services.py là tầng xử lý nghiệp vụ mà Controller gọi tới.
Thử nghiệm: chụp màn hình Postman trước và sau khi xếp lịch, và một trường hợp AI báo không đủ tín chỉ.
Tài liệu Claude API về tool use để tham khảo thêm: https://docs.claude.com/en/docs/agents-and-tools/tool-use/overview
